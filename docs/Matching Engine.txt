import re
from database import db

class ScholarshipBrain:
    def calculate_score(self, user_profile, scholarship):
        score = 0
        reasons = []

        if user_profile['major'] in scholarship['title'] or scholarship['category'] == '전공':
            score += 40
            reasons.append("전공 분야 일치 가능성 높음")

        if user_profile['location'][:2] in scholarship['title']:
            score += 30
            reasons.append(f"지역({user_profile['location']}) 제한 조건 충족")

        score += 10
        return {"score": score, "reasons": reasons}

    def get_matches(self, user_id):
        user = db.get_user_profile(user_id)
        if not user: return []
        all_scholarships = db.get_all_scholarships()
        matched_list = []
        for sch in all_scholarships:
            analysis = self.calculate_score(user, sch)
            if analysis['score'] >= 40:
                matched_list.append({
                    "title": sch['title'], "score": analysis['score'],
                    "reasons": analysis['reasons'], "link": sch['source'], "period": sch['period']
                })
        return sorted(matched_list, key=lambda x: x['score'], reverse=True)

brain = ScholarshipBrain()
