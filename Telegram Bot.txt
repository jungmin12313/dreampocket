import logging
from telegram import Update, ReplyKeyboardMarkup, ReplyKeyboardRemove
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters, ConversationHandler, ContextTypes
from database import db

GPA, INCOME, LOCATION, MAJOR = range(4)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    await update.message.reply_text("안녕하세요! 장학금 AI 비서 Antigravity입니다.🚀\nGPA를 입력해주세요 (예: 3.5/4.5)")
    return GPA

async def get_gpa(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    context.user_data['gpa'] = update.message.text
    await update.message.reply_text("소득 분위(1~10)를 입력해주세요.")
    return INCOME

async def get_income(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    context.user_data['income'] = update.message.text
    await update.message.reply_text("거주 지역을 입력해주세요.")
    return LOCATION

async def get_location(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    context.user_data['location'] = update.message.text
    await update.message.reply_text("전공을 입력해주세요.")
    return MAJOR

async def get_major(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    user_data = context.user_data
    user_data['major'] = update.message.text
    db.save_user_profile(str(update.effective_user.id), user_data)
    await update.message.reply_text("✅ 프로필 등록 완료!")
    return ConversationHandler.END
